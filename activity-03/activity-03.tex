% ================================================================================ %
%                                    Problem 01                                    %
% ================================================================================ %
\begin{problem}
Your \texttt{docker-compose.yml}.
\end{problem}

\begin{solution}
	Here's the \texttt{docker-compose.yml} file:
	\begin{codingbox*}[language=YAML]
name: activity-03

services:
  todo-webapp:
    build:
      context: ./src/todo-webapp
    container_name: webapp
    ports:
      - "3000:3000"
    stdin_open: true
    tty: true
    networks:
      - todo-net
    depends_on:
      - api-gateway

  todo:
    build:
      context: ./src/todo
    container_name: todo-service
    environment:
      REDIS_HOST: redis
      REDIS_PORT: "6379"
      NOTIFICATION_HOST: notification-service
      NOTIFICATION_PORT: "9000"
    networks:
      - todo-net
    depends_on:
      - redis
      - todo-notification

  todo-notification:
    build:
      context: ./src/todo-notification
    container_name: notification-service
    environment:
      REDIS_HOST: redis
      REDIS_PORT: "6379"
    networks:
      - todo-net
    depends_on:
      - redis

  api-gateway:
    image: nginx:alpine
    container_name: api-gateway
    ports:
      - "8000:8000"
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
    networks:
      - todo-net
    depends_on:
      - todo
      - todo-notification

  redis:
    image: redis:7-alpine
    container_name: redis
    networks:
      - todo-net

networks:
  todo-net:
    name: todo-net
    driver: bridge
\end{codingbox*}
\end{solution}
